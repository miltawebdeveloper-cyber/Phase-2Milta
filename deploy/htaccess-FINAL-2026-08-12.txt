##################################
# EXPOSED FILES — STOPGAP                     # <<< ADDED 2026-08-12
##################################
# https://www.miltafs.com/dist.zip returns 200 and downloads 198 MB today. That
# archive contains .env (Brevo + Supabase keys), server/.env, and full .git
# history. This block makes it un-downloadable within seconds of upload, which
# buys time — it does NOT make the file gone, and it does NOT un-leak anything
# already taken. Delete the file and rotate the keys.
#
# Scoped to archive/data extensions ONLY. Do not add json|js here: that would
# block /manifest.json and every fingerprinted bundle in /assets.
<FilesMatch "\.(zip|tar|gz|tgz|rar|7z|bak|sql|env|log|ini)$">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
    Deny from all
  </IfModule>
</FilesMatch>

# Dotfile directories. NOTE: <DirectoryMatch> and <Directory> are NOT permitted
# in .htaccess — Apache returns 500 "not allowed here" if you use them. That is
# why this is a RedirectMatch and not the DirectoryMatch block suggested in the
# older htaccess-patch.txt. Do not paste that one in.
RedirectMatch 404 "(?i)/\.(git|env|svn|hg|idea|vscode)(/|$)"

# Project directories that should not be in a web root at all. These are a
# stopgap for the same reason as above — delete the directories, then this
# block becomes dead weight and can come out.
RedirectMatch 404 "^/(src|scripts|server|node_modules|assets-originals)(/|$)"

##################################
# HSTS
##################################
# Tells browsers to refuse plain http for this domain. Deliberately starts at
# 5 minutes: a long max-age cannot be withdrawn quickly. After a week of clean
# https, raise it to:
#   Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
# Do NOT add `preload` unless every current and future subdomain is https-only.
<IfModule mod_headers.c>
  Header always set Strict-Transport-Security "max-age=300"
</IfModule>

# SOFT 404 — PATH-LEVEL FIXES                 # <<< ADDED 2026-08-12
# KEEP DIRECTLY ABOVE THE PRERENDERED-PAGES BLOCK
##################################
# Both rules below are conditional on what the build actually produced, so
# neither needs a hardcoded slug list and neither can strand a real page.

# /uk/blogs/<slug> for a post that exists only in the US table. blogs_uk holds
# two posts; anything else under /uk/blogs/ was a US post. Fires ONLY when the
# UK build has no page for that slug and the US build does — so when a UK post
# is added and rebuilt, the rule stops firing for it by itself.
# $1 in the conditions is the capture from the RewriteRule beneath them.
RewriteCond %{DOCUMENT_ROOT}/uk/blogs/$1.html !-f
RewriteCond %{DOCUMENT_ROOT}/us/blogs/$1.html -f
RewriteRule ^uk/blogs/([^/]+?)/?$ /us/blogs/$1 [R=301,L]

# Trailing slash on a route that was built as a flat <name>.html file.
# The !-d guard is what makes this safe: every slash-canonical URL on this site
# (/us/services/<x>/ and 129 others) is a real directory served by DirectoryIndex
# and is skipped here. Verified 2026-08-12 against all 130 slash URLs in the
# built sitemap — zero of them lack a directory, so zero are affected.
# Only fires when the directory does NOT exist and the flat file does.
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{DOCUMENT_ROOT}/$1.html -f
RewriteRule ^(.+?)/$ /$1 [R=301,L]

##################################
# PRERENDERED PAGES (KEEP DIRECTLY ABOVE THE SPA FALLBACK)
##################################
# `npm run build` writes static HTML for every route. Trailing-slash routes
# land as directories and already work via DirectoryIndex. Routes with NO
# trailing slash are written as <name>.html and need this rule — without it
# they fall through to the SPA fallback below, which is why /about, /contact,
# /blogs and ~80 other URLs currently return the HOME PAGE.
#
# There is deliberately NO `RewriteCond %{REQUEST_FILENAME} !-d` here.
# /uk and /career are both pages AND parents of child routes, so dist/ holds
# uk.html AND a uk/ directory. With `!-d` this rule would skip them and
# mod_dir's DirectorySlash would 301 /uk -> /uk/ and /career -> /career/,
# turning two canonical URLs into redirects. The `.html -f` test below is
# already precise enough on its own.
#
# No loop risk: once rewritten to /about.html the `!-f` condition is false on
# the next pass, so the rule cannot fire twice.
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{DOCUMENT_ROOT}%{REQUEST_URI}.html -f
RewriteRule ^(.*)$ /$1.html [L]

##################################
# REACT SPA ROUTING (KEEP LAST)
##################################
# Ignore existing files and directories
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d

# Ignore API backend routes specifically so they aren't caught by the SPA React Router
RewriteCond %{REQUEST_URI} !^/api/ [NC]

# Redirect all remaining non-file/non-directory requests to index.html
RewriteRule ^(.*)$ /index.html [QSA,L]
